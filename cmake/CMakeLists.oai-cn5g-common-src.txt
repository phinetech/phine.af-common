# Installation rules
################################################################################

# Install all libraries including those built in subdirectories
# We need to export all interdependent targets together

    EXPORT oai_cn5g_common_targets
    ARCHIVE DESTINATION lib
    LIBRARY DESTINATION lib
    RUNTIME DESTINATION bin
    INCLUDES DESTINATION include/oai
)

# Install header files preserving directory structure
# This allows relative includes within headers to work correctly
